cmake_minimum_required(VERSION 3.16)
project(power_manager_host_tests LANGUAGES CXX)
enable_testing()
add_library(pm_core ../core/pm_core.cpp ../core/pm_modbus.cpp)
target_include_directories(pm_core PUBLIC ../core/include)
target_compile_features(pm_core PUBLIC cxx_std_17)
target_compile_options(pm_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(pm_core_tests test_core.cpp)
target_link_libraries(pm_core_tests PRIVATE pm_core)
target_compile_options(pm_core_tests PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME pm_core_tests COMMAND pm_core_tests)
